  <script>

    const tg = window.Telegram?.WebApp;

    if (tg) { tg.ready(); tg.expand(); }


    const SUITS = ['♠', '♥', '♦', '♣'];

    const RANKS = ['6', '7', '8', '9', '10', 'J', 'Q', 'K', 'A'];

    const VALUES = { '6':6, '7':7, '8':8, '9':9, '10':10, 'J':11, 'Q':12, 'K':13, 'A':14 };


    let deck = [], playerHand = [], botHand = [];

    let trumpSuit = '', trumpCard = null;

    let table = []; // [{ attack, defend }]

    let isPlayerTurn = true;


    function initGame() {

      deck = [];

      for (let s of SUITS) {

        for (let r of RANKS) {

          deck.push({ suit: s, rank: r, isRed: s === '♥' || s === '♦' });

        }

      }

      deck.sort(() => Math.random() - 0.5);


      trumpCard = deck[deck.length - 1];

      trumpSuit = trumpCard.suit;


      playerHand = deck.splice(0, 6);

      botHand = deck.splice(0, 6);


      render();

    }


    function renderCard(c) {

      return `<div class="card ${c.isRed ? 'red' : ''}">${c.rank}<div>${c.suit}</div></div>`;

    }


    function render() {

      document.getElementById('deck-count').innerText = deck.length;

      document.getElementById('bot-count').innerText = botHand.length;

      document.getElementById('trump-val').innerText = `${trumpCard.rank}${trumpCard.suit}`;


      // Рука игрока

      const handDiv = document.getElementById('player-hand');

      handDiv.innerHTML = playerHand.map((c, i) => 

        `<div onclick="playerPlayCard(${i})">${renderCard(c)}</div>`

      ).join('');


      // Стол

      const tableDiv = document.getElementById('table');

      tableDiv.innerHTML = table.map(p => `

        <div class="card-pair">

          ${renderCard(p.attack)}

          ${p.defend ? renderCard(p.defend) : '<div style="height:40px"></div>'}

        </div>

      `).join('');


      // Кнопка

      const btn = document.getElementById('action-btn');

      if (isPlayerTurn) {

        btn.innerText = "Бито";

        btn.disabled = table.length === 0 || table.some(p => !p.defend);

      } else {

        btn.innerText = "Беру карту";

        btn.disabled = false;

      }

    }


    function playerPlayCard(idx) {

      const card = playerHand[idx];

      if (isPlayerTurn) {

        // Атака игрока

        if (table.length === 0 || table.some(p => p.attack.rank === card.rank || p.defend?.rank === card.rank)) {

          playerHand.splice(idx, 1);

          table.push({ attack: card, defend: null });

          render();

          setTimeout(botDefend, 600);

        }

      } else {

        // Защита игрока

        const undefended = table.find(p => !p.defend);

        if (undefended && canBeat(card, undefended.attack)) {

          playerHand.splice(idx, 1);

          undefended.defend = card;

          render();

          checkRoundEnd();

        }

      }

    }


    function canBeat(c1, c2) {

      if (c1.suit === c2.suit) return VALUES[c1.rank] > VALUES[c2.rank];

      return c1.suit === trumpSuit;

    }


    function botDefend() {

      const undefended = table.find(p => !p.defend);

      if (!undefended) return;


      const beatCardIdx = botHand.findIndex(c => canBeat(c, undefended.attack));

      if (beatCardIdx !== -1) {

        undefended.defend = botHand.splice(beatCardIdx, 1)[0];

        document.getElementById('status').innerText = "Бот отбился. Можешь подкинуть еще или нажать «Бито».";

      } else {

        document.getElementById('status').innerText = "Бот не может отбиться и забирает карты.";

        setTimeout(() => {

          table.forEach(p => { botHand.push(p.attack); if (p.defend) botHand.push(p.defend); });

          table = [];

          refillHands();

          render();

        }, 1200);

      }

      render();

    }


    function handleAction() {

      if (isPlayerTurn) {

        // Бито

        table = [];

        refillHands();

        isPlayerTurn = false;

        document.getElementById('status').innerText = "Ход бота. Отбивайся!";

        render();

        setTimeout(botAttack, 800);

      } else {

        // Взял карты

        table.forEach(p => { playerHand.push(p.attack); if (p.defend) playerHand.push(p.defend); });

        table = [];

        refillHands();

        document.getElementById('status').innerText = "Ты взял карты. Ход бота.";

        render();

        setTimeout(botAttack, 800);

      }

    }


    function botAttack() {

      if (botHand.length === 0) return;

      const card = botHand.shift();

      table.push({ attack: card, defend: null });

      render();

    }


    function refillHands() {

      while (playerHand.length < 6 && deck.length > 0) playerHand.push(deck.shift());

      while (botHand.length < 6 && deck.length > 0) botHand.push(deck.shift());

    }


    initGame();
